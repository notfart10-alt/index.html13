const colors = {
  Common:"#c1c8d2", Rare:"#50aaff", Epic:"#ce83ff",
  Legendary:"#ffb83f", Icon:"#ffe987"
};

let inventory = [];
let busy = false;

try {
  const saved = JSON.parse(localStorage.getItem("footballRNG") || "[]");
  if (Array.isArray(saved)) {
    inventory = saved.filter(p => players.some(x => x.name === p.name));
  }
} catch(e) {}

function save() {
  try {
    localStorage.setItem("footballRNG", JSON.stringify(inventory));
  } catch(e) {}
}

function pickPlayer() {
  const total = players.reduce((sum,p) => sum + p.weight, 0);
  let n = Math.random() * total;
  for (const p of players) {
    n -= p.weight;
    if (n <= 0) return p;
  }
  return players[0];
}

  const color = colors[p.rarity];
  return `

      <div class="club">${p.club}</div>
      <div class="stats">
        <div><b>PAC</b> ${Math.min(99,p.rating+2)}</div>
        <div><b>SHO</b> ${p.rating}</div>
        <div><b>SKL</b> ${Math.min(99,p.rating+1)}</div>
      </div>
    </div>`;
}

function renderInventory() {
  document.getElementById("count").textContent = inventory.length;
  const box = document.getElementById("inventory");

  if (!inventory.length) {
    box.innerHTML = '<div class="empty">⚽<br>Din samling är tom.<br>Tryck på RNG!</div>';
    return;
  }

  box.innerHTML = inventory.slice().reverse().map((p,i) => `
    <div class="mini" data-index="${inventory.length-1-i}">
      ${card(p)}
      <div class="mini-label">${p.name}</div>
    </div>`).join("");
}

function reveal(p) {
  document.getElementById("result").innerHTML = card(p);
  document.getElementById("result").classList.add("show");
  document.getElementById("name").textContent = p.name;
  document.getElementById("info").textContent =
    `${p.flag} ${p.club} · ${p.pos} · ${p.rating} OVR`;
  document.getElementById("label").textContent = "PLAYER UNLOCKED";
}

const wait = ms => new Promise(resolve => setTimeout(resolve,ms));

async function spin() {
  if (busy) return;
  busy = true;

  const btn = document.getElementById("spin");
  const stage = document.getElementById("stage");
  const winner = pickPlayer();
  const start = performance.now();
  const duration = 3500;

  btn.disabled = true;
  btn.textContent = "⏳ SPINNING...";
  document.getElementById("notice").textContent = "";
  document.getElementById("result").classList.remove("show");
  document.getElementById("label").textContent = "SEARCHING FOR PLAYER...";
  stage.classList.add("spinning");

  let last = 0;

  await new Promise(resolve => {
    function frame(now) {
      const progress = Math.min((now-start)/duration,1);
      const delay = 35 + progress*progress*400;

      if (now-last > delay || progress === 1) {
        const p = progress > .90
          ? winner
          : players[Math.floor(Math.random()*players.length)];

        document.getElementById("name").textContent = p.name;
        document.getElementById("info").textContent =
          progress > .90 ? "Almost there..." : "Rolling...";
        last = now;
      }

      if (progress < 1) requestAnimationFrame(frame);
      else resolve();
    }
    requestAnimationFrame(frame);
  });

  stage.classList.remove("spinning");
  inventory.push(winner);
  save();
  renderInventory();
  reveal(winner);

  const messages = {
    Common:"PLAYER ADDED TO YOUR COLLECTION!",
    Rare:"💎 NICE! RARE PLAYER!",
    Epic:"💜 HUGE PULL! EPIC PLAYER!",
    Legendary:"🔥 INCREDIBLE! LEGENDARY!",
    Icon:"👑 UNBELIEVABLE! YOU PACKED AN ICON!"
  };

  document.getElementById("notice").textContent = messages[winner.rarity];
  btn.textContent = "⚡ SPIN AGAIN ⚡";
  btn.disabled = false;
  busy = false;
}

document.getElementById("spin").addEventListener("click",spin);

  if (p) {

    document.getElementById("notice").textContent = "PLAYER FROM YOUR INVENTORY";
    window.scrollTo({top:0,behavior:"smooth"});
  }
});

  }
});
